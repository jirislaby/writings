\section{Configuration: Kconfig}
\frame{\sectionpage}

\begin{frame}{Why Configure?}

  \begin{itemize}[<+->]
    \item One source tree
      \begin{itemize}[<.->]
	\item 20+ architectures
	\item Phones, servers, routers, cars, mainframes
      \end{itemize}
    \item Each option is
      \begin{itemize}[<.->]
	\item \cmd{y} -- built in the kernel image (\file{vmlinux})
	\item \cmd{m} -- built as a module (\file{.ko})
	\item \cmd{n} -- not built at all
      \end{itemize}
    \item Options depend on each other
      \begin{itemize}[<.->]
	\item No USB storage without USB
	\item ext4 needs jbd2
      \end{itemize}
    \item Described in \file{Kconfig} files
      \begin{itemize}[<.->]
	\item Own language: \doc{Documentation/kbuild/kconfig-language.rst}
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}{Configuration Targets}

  \begin{itemize}[<+->]
    \item \heading{Start from something}
      \begin{itemize}[<+->]
	\item \cmd{make defconfig} -- arch default (\file{arch/x86/configs/x86_64_defconfig})
	\item \cmd{make localmodconfig} -- only modules loaded now (\cmd{lsmod})
	\item \cmd{make allmodconfig}, \cmd{allyesconfig}, \cmd{tinyconfig}, \cmd{randconfig}
	\item Distro config: \file{/proc/config.gz}, \file{/boot/config-*}
      \end{itemize}
    \item \heading{Modify}
      \begin{itemize}[<+->]
	\item \cmd{make menuconfig} (ncurses), \cmd{nconfig}, \cmd{xconfig} (Qt)
	\item \cmd{scripts/config -e EXT4_FS -m USB_STORAGE -d DRM}
	%\item Fragments: \cmd{make defconfig kvm_guest.config}
      \end{itemize}
    \item \heading{Update after a kernel upgrade}
      \begin{itemize}[<+->]
	\item \cmd{make oldconfig} -- ask for new options
	\item \cmd{make olddefconfig} -- use defaults for new options
	\item \cmd{make listnewconfig}
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Configuration Outputs}

  \begin{itemize}[<+->]
    \item \file{.config} -- what you edit (via tools)
      \begin{lstlisting}[language={}]
CONFIG_EXT4_FS=m
CONFIG_EXT4_FS_POSIX_ACL=y
# CONFIG_EXT4_DEBUG is not set
      \end{lstlisting}
    \item \cmd{make syncconfig} (run automatically) generates
      \begin{itemize}[<+->]
	\item \file{include/config/auto.conf} -- for \emph{make} (Kbuild)
	\item \file{include/generated/autoconf.h} -- for \emph{C} (\cmd{-include})
	  \begin{lstlisting}
#define CONFIG_EXT4_FS_MODULE 1
#define CONFIG_EXT4_FS_POSIX_ACL 1
	  \end{lstlisting}
	\item \file{include/config/EXT4_FS}, \dots -- one (empty) file per symbol
	  \begin{itemize}
	    \item Timestamps! More in \emph{Incremental Builds}
	  \end{itemize}
      \end{itemize}
  \end{itemize}

\end{frame}
