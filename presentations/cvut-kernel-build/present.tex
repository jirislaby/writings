\begin{frame}{Why Configure?}

  \begin{itemize}[<+->]
    \item One source tree
      \begin{itemize}[<.->]
	\item 20+ architectures
	\item Phones, servers, routers, cars, mainframes
      \end{itemize}
    \item Each option is
      \begin{itemize}[<.->]
	\item \cmd{y} -- built in the kernel image (\file{vmlinux})
	\item \cmd{m} -- built as a module (\file{.ko})
	\item \cmd{n} -- not built at all
      \end{itemize}
    \item Options depend on each other
      \begin{itemize}[<.->]
	\item No USB storage without USB
	\item ext4 needs jbd2
      \end{itemize}
    \item Described in \file{Kconfig} files
      \begin{itemize}[<.->]
	\item Own language: \file{Documentation/kbuild/kconfig-language.rst}
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Kconfig Language}{Real example: \file{fs/ext4/Kconfig}}

config EXT4_FS

\begin{frame}{Kconfig Language}{Keywords}

  \begin{itemize}[<+->]
    \item Types: \cmd{bool}, \cmd{tristate}, \cmd{int}, \cmd{hex}, \cmd{string}
    \item \cmd{depends on X}
      \begin{itemize}[<.->]
	\item Option not visible unless \cmd{X} is set
	\item \cmd{m} limits dependents to \cmd{m}
      \end{itemize}
    \item \cmd{select X}
      \begin{itemize}[<.->]
	\item Forces \cmd{X} on, \emph{ignoring} \cmd{X}'s dependencies
	\item Use for library-like options only
      \end{itemize}
    \item \cmd{imply X} -- weak select (user can still say \cmd{n})
    \item \cmd{default}, \cmd{help}, \cmd{prompt}
    \item Structure: \cmd{menu}, \cmd{choice}, \cmd{if}, \cmd{source}
      \begin{itemize}[<.->]
	\item \file{arch/x86/Kconfig} $\rightarrow$ \cmd{source "fs/Kconfig"}
	  $\rightarrow$ \cmd{source "fs/ext4/Kconfig"} $\rightarrow$ \dots
      \end{itemize}
  \end{itemize}

\end{frame}

\section{Kbuild}
\frame{\sectionpage}

\begin{frame}[fragile]{Kbuild Files}{Real example: \file{fs/ext4/Makefile}}

  \begin{lstlisting}[language=make,basicstyle=\footnotesize\ttfamily]
obj-$(CONFIG_EXT4_FS) += ext4.o

ext4-y  := balloc.o bitmap.o block_validity.o dir.o ext4_jbd2.o extents.o \
           ... super.o symlink.o sysfs.o xattr.o xattr_hurd.o ...

ext4-$(CONFIG_EXT4_FS_POSIX_ACL)  += acl.o
ext4-$(CONFIG_EXT4_FS_SECURITY)   += xattr_security.o
ext4-$(CONFIG_FS_VERITY)          += verity.o
  \end{lstlisting}

  \pause

  \begin{itemize}[<+->]
    \item \file{CONFIG_EXT4_FS=y} $\rightarrow$ \cmd{obj-y += ext4.o} $\rightarrow$ into \file{vmlinux}
    \item \file{CONFIG_EXT4_FS=m} $\rightarrow$ \cmd{obj-m += ext4.o} $\rightarrow$ \file{ext4.ko}
    \item \file{CONFIG_EXT4_FS} unset $\rightarrow$ \cmd{obj- += ext4.o} $\rightarrow$ ignored
    \item \cmd{ext4-y} -- \emph{composite} object: \file{ext4.o} is linked from these
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Kbuild Files}{Syntax Summary}

  \begin{itemize}[<+->]
    \item Named \file{Kbuild} or \file{Makefile} (\file{Kbuild} preferred if both)
    \item Descending into subdirectories (\file{fs/Makefile})
      \begin{lstlisting}[language=make]
obj-$(CONFIG_EXT4_FS)    += ext4/
obj-$(CONFIG_BTRFS_FS)   += btrfs/
      \end{lstlisting}
    \item Flags
      \begin{lstlisting}[language=make]
ccflags-y                := -I$(src)/include   # whole directory
CFLAGS_super.o           := -DDEBUG            # single file
      \end{lstlisting}
    \item Not \emph{normal} Makefiles: no rules, only variables
      \begin{itemize}[<.->]
	\item Rules live in \file{scripts/Makefile.build}
	\item Docs: \file{Documentation/kbuild/makefiles.rst}
      \end{itemize}
    \item Similar to \file{meson.build}, \file{CMakeLists.txt}: \emph{what}, not \emph{how}
  \end{itemize}

\end{frame}

\begin{frame}{Kbuild}{How It Walks the Tree}

  \begin{itemize}[<+->]
    \item Top-level \file{Makefile}
      \begin{itemize}[<.->]
	\item Includes \file{arch/\$(SRCARCH)/Makefile} (flags, boot targets)
	\item Starts with top-level dirs: \file{init/}, \file{kernel/}, \file{mm/},
	  \file{fs/}, \file{drivers/}, \dots
      \end{itemize}
    \item For each directory: \cmd{make -f scripts/Makefile.build obj=fs/ext4}
      \begin{itemize}[<.->]
	\item Reads \file{fs/ext4/Makefile}
	\item Compiles \cmd{obj-y} and \cmd{obj-m} objects
	\item Descends into subdirectories (recursive make)
	\item Collects \cmd{obj-y} into \file{fs/ext4/built-in.a}
	\item Records \cmd{obj-m} into \file{modules.order}
      \end{itemize}
    \item Parallel (\cmd{-j}) across directories
  \end{itemize}

\end{frame}

\section{Building and Linking the Kernel}
\frame{\sectionpage}

\begin{frame}[fragile]{Compiling One Object}

  \heading{\cmd{make V=1 fs/ext4/dir.o} (shortened)}
  \begin{lstlisting}[language=bash]
gcc -Wp,-MMD,fs/ext4/.dir.o.d -nostdinc -I./arch/x86/include -I./include \
  -include ./include/linux/compiler-version.h -include ./include/linux/kconfig.h \
  -include ./include/linux/compiler_types.h -D__KERNEL__ -std=gnu11 \
  -fno-strict-aliasing -mno-sse -mno-red-zone -mcmodel=kernel -O2 -g ... \
  -DKBUILD_MODFILE='"fs/ext4/ext4"' -DKBUILD_BASENAME='"dir"' \
  -DKBUILD_MODNAME='"ext4"' -DMODULE -c -o fs/ext4/dir.o fs/ext4/dir.c
  \end{lstlisting}

  \pause

  \begin{itemize}[<+->]
    \item \cmd{-nostdinc} -- no libc, no \file{/usr/include}
    \item \file{kconfig.h} includes \file{autoconf.h} -- every file sees all
      \file{CONFIG_*}
    \item \cmd{-mno-sse} -- no FPU in the kernel
    \item \cmd{-MMD} -- dependency file (later)
    \item \cmd{-DMODULE} -- built as \cmd{m}
  \end{itemize}

\end{frame}

\begin{frame}{From Objects to \file{vmlinux}}

  \begin{itemize}[<+->]
    \item \file{dir/built-in.a} -- per directory
      \begin{itemize}[<.->]
	\item \emph{Thin} archive: references \file{.o} files, no copies
	\item Historically \file{built-in.o} (\cmd{ld -r}), until 4.13
      \end{itemize}
    \item \file{vmlinux.a} -- all top-level \file{built-in.a} together
    \item \file{vmlinux.o} -- \cmd{ld -r} of everything
      \begin{itemize}[<.->]
	\item \cmd{objtool} -- validates stack frames, generates ORC unwind
	  data, \dots
	\item \cmd{modpost} -- section mismatches (\file{.init} vs.\ the rest)
      \end{itemize}
    \item \file{scripts/link-vmlinux.sh}
      \begin{itemize}[<.->]
	\item Links vmlinux \emph{several times}. Why?
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}{Why Link \file{vmlinux} Several Times?}

  \begin{itemize}[<+->]
    \item \emph{kallsyms}: the kernel contains its own symbol table
      \begin{itemize}[<.->]
	\item Stack traces with function names, \file{/proc/kallsyms}
      \end{itemize}
    \item Chicken \& egg
      \begin{itemize}[<.->]
	\item Symbol table needs final addresses
	\item Adding the table moves the addresses
      \end{itemize}
    \item The solution
      \begin{enumerate}[<.->]
	\setcounter{enumi}{-1}
	\item Empty table
	\item \file{.tmp_vmlinux1} $\rightarrow$ table of the right \emph{size}
	\item \file{.tmp_vmlinux2} $\rightarrow$ table with the right \emph{addresses}
	\item \file{.tmp_vmlinux3} -- only if something still moved
	\item Final link with the correct table
      \end{enumerate}
    %\item BTF is generated from \file{.tmp_vmlinux1} too (later)
    \item \file{System.map} must match the embedded table (checked)
  \end{itemize}

\end{frame}

\section{ELF and DWARF in a Nutshell}
\frame{\sectionpage}

\begin{frame}{ELF}

  \begin{columns}[c]
    \begin{column}{.35\textwidth}
      \tikzstyle{pre} = [<-,>=stealth',semithick]
      \tikzstyle{sec} = [draw,semithick,minimum width=3.5cm, minimum height=6mm]
      \begin{tikzpicture}[node distance=1pt, font=\scriptsize]
	\node [sec] (HEAD) {ELF header};
	\node [sec,below=of HEAD] (PRGHDR) {Program Header Table};
	\node [sec,below=of PRGHDR] (TEXT) {.text};
	\node [sec,below=of TEXT] (RODATA) {.rodata};
	\node [sec,below=of RODATA] (DATA) {.data};
	\node [minimum width=3.5cm,below=of DATA] (MORE) {\dots};
	\node [sec,below=of MORE] (DEBUG) {.debug\_info, \dots};
	\node [sec,below=of DEBUG] (SECHDR) {Section Header Table};
      \end{tikzpicture}
    \end{column}
    \begin{column}{.6\textwidth}
      \heading{Executable and Linkable Format}
      \begin{itemize}[<+->]
	\item Object files, executables, libraries, core dumps
	\item \file{.o}, \file{.ko}, \file{vmlinux} -- all ELF
	\item \emph{Sections}: code, data, symbols, relocations, debug info
	\item Kernel-specific sections
	  \begin{itemize}[<.->]
	    \item \file{.init.text} -- freed after boot
	    \item \file{__ksymtab} -- exported symbols
	    \item \file{__ex_table} -- exception fixups
	    \item \file{.modinfo} -- module metadata
	  \end{itemize}
	\item Tools: \cmd{readelf}, \cmd{objdump}, \cmd{nm}, \cmd{objcopy}
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{DWARF}

  \heading{Debugging information format}
  \begin{itemize}[<+->]
    \item Maps machine code back to the source
      \begin{itemize}[<.->]
	\item Addresses $\leftrightarrow$ files and lines
	\item Types, structures, variables, inlined functions, \dots
      \end{itemize}
    \item Stored in ELF sections \file{.debug_*}
    \item \cmd{CONFIG_DEBUG_INFO_DWARF5} (or \cmd{_DWARF4})
      \begin{itemize}[<.->]
	\item \emph{Huge}: \file{vmlinux} grows from tens to hundreds of MB
	\item Slower build
	\item Distributions ship it separately (\file{-debuginfo} packages)
      \end{itemize}
    \item Consumers
      \begin{itemize}[<.->]
	\item \cmd{gdb}, \cmd{addr2line}, \cmd{crash} (kernel dumps), \cmd{perf}
	\item \cmd{scripts/faddr2line}, \cmd{scripts/decode_stacktrace.sh}
	%\item \cmd{pahole} -- and the BTF generation!
      \end{itemize}
  \end{itemize}

\end{frame}

        struct list_head *         next;                 /*     0     8 */

        /* size: 16, cachelines: 1, members: 2 */
        /* last cacheline: 16 bytes */
};
  \end{lstlisting}

\end{frame}

\section{Incremental Builds}
\frame{\sectionpage}

\begin{frame}{The Problem}

  \begin{itemize}[<+->]
    \item Full build: minutes to hours
    \item Change one line $\Rightarrow$ rebuild \emph{only} what is needed
    \item \cmd{make}: rebuild a target if a prerequisite is newer
      \begin{itemize}[<.->]
	\item \cmd{gcc -MMD} gives all included headers
      \end{itemize}
    \item But
      \begin{enumerate}[<.->]
	\item \emph{Every} file includes \file{autoconf.h}
	  \begin{itemize}
	    \item Toggle one option $\Rightarrow$ rebuild everything!?
	  \end{itemize}
	\item Changed \emph{flags} (\cmd{ccflags-y}, \cmd{make W=1}, new
	  compiler) do not change any timestamp
	  \begin{itemize}
	    \item $\Rightarrow$ nothing rebuilt!?
	  \end{itemize}
      \end{enumerate}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{The Solution: \cmd{fixdep}}

  \begin{itemize}[<+->]
    \item \file{scripts/basic/fixdep} post-processes \file{.dir.o.d} into \file{.dir.o.cmd}
      \begin{lstlisting}[language=make]
savedcmd_fs/ext4/dir.o := gcc -Wp,-MMD,fs/ext4/.dir.o.d -nostdinc ...

deps_fs/ext4/dir.o := \
  fs/ext4/dir.c \
    $(wildcard include/config/UNICODE) \
    $(wildcard include/config/EXT4_FS_ENCRYPTION) \
  include/linux/fs.h \
  ...
      \end{lstlisting}
    \item[1.] \file{autoconf.h} dependency \emph{replaced} by \file{include/config/<SYMBOL>}
      \begin{itemize}[<.->]
	\item Only for symbols the file (or its headers) mention
	\item \cmd{syncconfig} touches only files of \emph{changed} symbols
      \end{itemize}
    \item[2.] \cmd{savedcmd_*} compared to the current command (\cmd{if_changed})
      \begin{itemize}[<.->]
	\item Different flags $\Rightarrow$ rebuild
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}{What Gets Rebuilt?}

  \begin{itemize}[<+->]
    \item \heading{\cmd{touch fs/ext4/dir.c}}
      \begin{itemize}[<.->]
	\item \file{dir.o} $\rightarrow$ \file{ext4.o} $\rightarrow$ \file{ext4.ko}
	  (if \cmd{m})
	\item \file{built-in.a} $\rightarrow$ \file{vmlinux} relink (if \cmd{y})
      \end{itemize}
    \item \heading{\cmd{scripts/config -e EXT4_DEBUG}}
      \begin{itemize}[<.->]
	\item Only files referencing \file{CONFIG_EXT4_DEBUG}
      \end{itemize}
    \item \heading{\cmd{touch include/linux/fs.h}}
      \begin{itemize}[<.->]
	\item Thousands of files
	\item Header hygiene matters!
      \end{itemize}
    \item \heading{\cmd{make W=1}} -- different flags $\Rightarrow$ everything
    \item \heading{Why was it rebuilt?}
      \begin{itemize}[<.->]
	\item \cmd{make V=2} prints the reason
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}{Faster Builds}

  \begin{itemize}[<+->]
    \item \cmd{make -j\$(nproc)} -- obviously
    \item Smaller config: \cmd{localmodconfig}
    \item Build only what you touch: \cmd{make fs/ext4/}, \cmd{make M=fs/ext4}
    \item \cmd{O=} -- separate build dirs per config/arch, one source tree
    \item \cmd{ccache} -- caches compiler outputs
      \begin{itemize}[<.->]
	\item \cmd{make CC='ccache gcc'}
	\item Helps when switching branches back and forth
      \end{itemize}
    \item Distributed: \cmd{icecream}, \cmd{distcc}
    \item Skip debug info when not needed
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Incremental Builds}

  \demo{What gets rebuilt}

  \begin{lstlisting}[language=bash]
make -j$(nproc)                       # nothing to do
touch fs/ext4/dir.c && make V=2 -j$(nproc)
cat fs/ext4/.dir.o.cmd
scripts/config -e EXT4_DEBUG && make -j$(nproc)
ls -l --time-style=full-iso include/config/EXT4_DEBUG include/config/EXT4_FS
  \end{lstlisting}

\end{frame}

\section{Modules}
\frame{\sectionpage}

\begin{frame}{Kernel Modules}

  \begin{itemize}[<+->]
    \item \file{.ko} = relocatable ELF (like \file{.o}) + metadata
    \item Loaded at runtime: \cmd{insmod}, \cmd{modprobe} (+ dependencies)
      \begin{itemize}[<.->]
	\item The kernel itself links it (resolves symbols, relocates)
	\item Only \cmd{EXPORT_SYMBOL()}-ed symbols available
      \end{itemize}
    \item \emph{In-tree}: \cmd{obj-m} in the kernel's Kbuild files
    \item \emph{Out-of-tree}: your own directory
      \begin{itemize}[<.->]
	\item Needs a \emph{configured and prepared} kernel tree
	\item \file{/lib/modules/\$(uname -r)/build}
	\item Distros: \file{kernel-devel} + \file{kernel-default-devel}, \file{linux-headers-*}
      \end{itemize}
    \item Must match the running kernel
      \begin{itemize}[<.->]
	\item \emph{vermagic}: version, SMP, preempt, \dots
	\item \cmd{CONFIG_MODVERSIONS}: CRC of each exported symbol's type
	  \begin{itemize}
	    \item SL
	  \end{itemize}
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Writing Your Own Module}{\file{hello.c}}

  \begin{lstlisting}[basicstyle=\scriptsize\ttfamily]
// SPDX-License-Identifier: GPL-2.0
#include <linux/init.h>
#include <linux/module.h>

static int __init hello_init(void)
{
	pr_info("Hello from %s\n", KBUILD_MODNAME);
	return 0;
}

static void __exit hello_exit(void)
{
	pr_info("Goodbye\n");
}

module_init(hello_init);
module_exit(hello_exit);
MODULE_DESCRIPTION("Hello world module");
MODULE_LICENSE("GPL");
  \end{lstlisting}

\end{frame}

\begin{frame}[fragile]{Writing Your Own Module}{\file{Makefile}}

  \begin{block}{\file{Makefile} -- just a wrapper}
    \begin{lstlisting}[language=make,basicstyle=\footnotesize\ttfamily]
KDIR ?= /lib/modules/$(shell uname -r)/build

obj-m := hello.o

all:
	$(MAKE) -C $(KDIR) M=$(CURDIR)

clean:
	$(MAKE) -C $(KDIR) M=$(CURDIR) clean
    \end{lstlisting}
  \end{block}

  \pause

  \begin{itemize}
    \item \cmd{-C} -- run the \emph{kernel's} Makefile
    \item \cmd{M=} -- but build the module in this directory
    \item Docs: \doc{Documentation/kbuild/modules.rst}
  \end{itemize}

\end{frame}

\begin{frame}{Linking a Module}

  \begin{center}
  \begin{tikzpicture}[node distance=5mm and 9mm]
    \node[bnode] (c) {hello.c};
    \node[bnode, right=of c] (o) {hello.o};
    \node<2->[bnode, below=of o] (modc) {hello.mod.c};
    \node<2->[bnode, right=of modc] (modo) {hello.mod.o};
    \node<2->[bnode, left=of modc] (sym) {Module.symvers\\(from kernel)};
    \node<3->[bnode, right=26mm of o] (ko) {hello.ko};
    \node<4->[bnode, right=of ko, fill=green!20] (kos) {hello.ko\\+signature};

    \draw[barrow] (c) -- node[btool, above] {gcc -c} (o);
    \draw<2->[barrow] (o) -- node[btool, right] {modpost} (modc);
    \draw<2->[barrow] (sym) -- (modc);
    \draw<2->[barrow] (modc) -- node[btool, above] {gcc -c} (modo);
    \draw<3->[barrow] (o) -- node[btool, above] {ld -r} (ko);
    \draw<3->[barrow] (modo) -- (ko);
    \draw<4->[barrow] (ko) -- node[btool, above] {sign-file} (kos);
  \end{tikzpicture}
  \end{center}

  \begin{itemize}[<+->]
    \item \cmd{gcc} -- same as for built-in, plus \cmd{-DMODULE}
    \item \cmd{modpost}
      \begin{itemize}
	\item checks undefined symbols against \file{Module.symvers}
	\item generates \file{.mod.c}: vermagic, dependencies, \file{struct module}
      \end{itemize}
    \item \cmd{ld -r} -- \file{.ko} stays relocatable
    \item Optional: signature (\cmd{CONFIG_MODULE_SIG}) -- at \cmd{modules_install}
  \end{itemize}

\end{frame}

sudo insmod hello.ko && sudo dmesg | tail -1

\section{How We Do It at SUSE}
\frame{\sectionpage}

\begin{frame}{Distribution Kernel}

  \begin{itemize}[<+->]
    \item One kernel for \emph{all} users and hardware
      \begin{itemize}[<.->]
	\item Nearly everything as modules (thousands of \file{.ko})
	\item Several archs: \xesh, arm64, ppc64le, s390x, \dots
	\item Several \emph{flavors}: \file{default}, \file{rt}, \file{64kb}, \file{kvmsmall}, \dots
      \end{itemize}
    \item Reproducible: anyone can rebuild the exact same kernel
    \item Built in an isolated environment (no network, clean chroot/VM)
      \begin{itemize}[<.->]
	\item Open Build Service (OBS): \url{https://build.opensuse.org/project/subprojects/Kernel}
      \end{itemize}
    \item Tumbleweed: latest upstream stable + a few patches
    \item SLE: older base + thousands of backported patches, supported for 10+ years
  \end{itemize}

\end{frame}

\begin{frame}{\file{kernel-source.git}}

  \begin{itemize}[<+->]
    \item \emph{Not} a kernel tree! A tree of \emph{patches}
      \begin{itemize}[<.->]
	\item \url{https://github.com/SUSE/kernel-source}
	\item One branch per product: \file{stable} (Tumbleweed), \file{SL-16.1},
	  \file{SLE15-SP7}, \dots
      \end{itemize}
    \item Content
      \begin{itemize}[<.->]
	\item \file{patches.*/} + \file{series.conf} -- patches on top of upstream tarball
	\item \file{config/<arch>/<flavor>} -- \file{.config} for each build
	\item \file{supported.conf} -- support status of modules
	\item \file{rpm/*.spec.in} -- templates of RPM spec files
	\item \file{scripts/} -- tooling
      \end{itemize}
    \item \cmd{scripts/sequence-patch} -- apply patches, get a kernel tree
    \item \cmd{scripts/tar-up} -- create package sources for OBS
  \end{itemize}

\end{frame}
