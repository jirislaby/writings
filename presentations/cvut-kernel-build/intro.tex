\begin{frame}{Tools Involved}

  \begin{itemize}[<+->]
    \item \cmd{make} -- drives everything
      \begin{itemize}[<.->]
	\item Kbuild: a (huge) set of Makefiles in \file{scripts/}
	\item You do \emph{not} need to understand them
      \end{itemize}
    \item Compiler: \cmd{gcc} or \cmd{clang} (\cmd{make LLVM=1})
    \item Binutils: \cmd{ld}/\cmd{ld.lld}, \cmd{objcopy}, \cmd{nm}, \cmd{readelf}, \dots
    \item Host tools built on the fly: \cmd{fixdep}, \cmd{kallsyms}, \cmd{modpost},
      \cmd{objtool}, \cmd{conf}, \dots
    \item Optional: \cmd{pahole} (BTF), \cmd{openssl} (signing), \cmd{rustc}, \dots
    \item Minimal versions: \doc{Documentation/process/changes.rst}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{Basic Invocation}

  \begin{itemize}[<+->]
    \item \cmd{make help} -- list of all targets (really useful!)
    \item \cmd{make defconfig} -- create \file{.config}
    \item \cmd{make -j\$(nproc)} -- build \file{vmlinux}, \file{bzImage} and modules
    \item \cmd{make -j\$(nproc) V=1} -- show the full commands
      \iffalse
    \item \cmd{make O=../build-x86} -- build out of the source tree
    \item \cmd{make ARCH=arm64 CROSS_COMPILE=aarch64-linux-gnu-} -- cross build
    \item \cmd{make LLVM=1 ...} -- \file{clang} + \file{lld} + \file{llvm-*} tools
      \fi
    \item \cmd{make fs/ext4/} -- build a single directory
    \item \cmd{make fs/ext4/dir.o} -- \ldots\ or a single object (\file{.i},
      \file{.s}, \file{.lst} too)
  \end{itemize}

\end{frame}
