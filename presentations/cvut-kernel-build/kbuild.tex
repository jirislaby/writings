\section{Kbuild}
\frame{\sectionpage}

\begin{frame}[fragile]{Kbuild Files}{Real example: \file{fs/ext4/Makefile}}

  \begin{columns}
    \begin{column}{.49\textwidth}
      \heading{\cmd{Kbuild} (priority over \file{Makefile})}
      \begin{lstlisting}[language=make,basicstyle=\footnotesize\ttfamily]
obj-y                   += fs/
      \end{lstlisting}
    \end{column}

    \pause

    \begin{column}{.49\textwidth}
      \heading{\cmd{fs/Makefile}}
      \begin{lstlisting}[language=make,basicstyle=\footnotesize\ttfamily]
obj-$(CONFIG_EXT4_FS)    += ext4/
      \end{lstlisting}
    \end{column}
  \end{columns}

  \pause

  \heading{\cmd{fs/ext4/Makefile}}
  \begin{lstlisting}[language=make,basicstyle=\footnotesize\ttfamily]
obj-$(CONFIG_EXT4_FS) += ext4.o

ext4-y  := balloc.o bitmap.o block_validity.o dir.o ext4_jbd2.o extents.o \
           ... super.o symlink.o sysfs.o xattr.o xattr_hurd.o ...

ext4-$(CONFIG_EXT4_FS_POSIX_ACL)  += acl.o
ext4-$(CONFIG_EXT4_FS_SECURITY)   += xattr_security.o
  \end{lstlisting}

  \pause

  \begin{itemize}[<+->]
    \item \file{CONFIG_EXT4_FS=y} $\rightarrow$ \cmd{obj-y += ext4.o} $\rightarrow$ into \file{vmlinux}
    \item \file{CONFIG_EXT4_FS=m} $\rightarrow$ \cmd{obj-m += ext4.o} $\rightarrow$ \file{ext4.ko}
    \item \file{CONFIG_EXT4_FS} unset $\rightarrow$ \cmd{obj- += ext4.o} $\rightarrow$ ignored
  \end{itemize}

\end{frame}

\begin{frame}{Kbuild Files}{Syntax Summary}

  \begin{itemize}[<+->]
    \item Not \emph{normal} Makefiles: no rules, only variables
    \item Rules live in \file{scripts/Makefile.build}
    \item Docs: \doc{Documentation/kbuild/makefiles.rst}
  \end{itemize}

\end{frame}

\begin{frame}{Kbuild}{How It Walks the Tree}

  \begin{itemize}[<+->]
    \item Top-level \file{Makefile} + \file{Kbuild}
      \begin{itemize}[<+->]
	\item Includes \file{arch/\$(SRCARCH)/Makefile} (flags, boot targets)
	\item Starts with top-level dirs: \file{init/}, \file{kernel/}, \file{mm/},
	  \file{fs/}, \file{drivers/}, \dots
      \end{itemize}
    \item For each directory: \cmd{make -f scripts/Makefile.build obj=<directory>}
      \begin{itemize}[<+->]
	\item For \file{<directory>} == \file{fs/ext4/}:
	\item Reads \file{fs/ext4/Makefile}
	\item Compiles \cmd{obj-y} and \cmd{obj-m} objects
	\item Descends into subdirectories (recursive make)
	\item Collects \cmd{obj-y} into \file{fs/ext4/built-in.a}
	\item Records \cmd{obj-m} into \file{modules.order}
      \end{itemize}
    \item Parallel (\cmd{-j}) across directories
  \end{itemize}

\end{frame}
