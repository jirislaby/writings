    \begin{column}{.6\textwidth}
      \heading{Executable and Linkable Format}
      \begin{itemize}[<+->]
	\item Object files, executables, libraries, core dumps
	\item \file{.o}, \file{.ko}, \file{vmlinux} -- all ELF
	\item \emph{Sections}: code, data, symbols, debug info, \dots
	\item Kernel-specific sections
	  \begin{itemize}[<.->]
	    \item \file{.init.\{data,text\}} -- freed after boot
	    \item \file{__ksymtab} -- exported symbols
	    \item \file{.modinfo} -- module metadata
	  \end{itemize}
	\item Tools: \cmd{readelf}, \cmd{objdump}, \cmd{nm}, \cmd{objcopy}
      \end{itemize}
    \end{column}
  \end{columns}

\begin{frame}[fragile]{ELF}{Examples}

  \begin{block}{Common Commands}
  \begin{lstlisting}[language=bash]
readelf -h vmlinux                      # header: type, arch, entry point
readelf -S fs/ext4/ext4.ko              # sections
readelf -l fs/ext4/ext4.ko              # program headers
readelf -s fs/ext4/ext4.ko | grep UND   # undefined symbols (from vmlinux)
objdump -d --no-show-raw-insn fs/ext4/dir.o | less
nm vmlinux | grep ' T start_kernel'
grep start_kernel System.map
modinfo fs/ext4/ext4.ko                 # parses .modinfo
  \end{lstlisting}
  \end{block}

\end{frame}

\begin{frame}{DWARF}

  \heading{Debugging information format}
  \begin{itemize}[<+->]
    \item Maps machine code back to the source
      \begin{itemize}[<.->]
	\item Addresses $\leftrightarrow$ files and lines
	\item Types, structures, variables, inlined functions, \dots
      \end{itemize}
    \item Stored in ELF sections \file{.debug_*}
    \item \cmd{CONFIG_DEBUG_INFO_DWARF5} (or \cmd{_DWARF4})
      \begin{itemize}[<.->]
	\item \emph{Huge}: \file{vmlinux} grows from tens to hundreds of MB
	\item Slower build
	\item Distributions ship it separately (\file{-debuginfo} packages)
      \end{itemize}
    \item Consumers
      \begin{itemize}[<.->]
	\item \cmd{gdb}, \cmd{addr2line}, \cmd{crash} (kernel dumps), \cmd{perf}
	\item \cmd{scripts/faddr2line}, \cmd{scripts/decode_stacktrace.sh}
	%\item \cmd{pahole} -- and the BTF generation!
      \end{itemize}
  \end{itemize}

\end{frame}
